<body>
    <h1>⚙️ Kendi Baret Logomu Tasarlıyorum</h1>
    <p>Yandaki kodlarda tekrarSayisi değişkeni değerini değiştir ve denemeler yap</p>

    <canvas id="gameCanvas" width="400" height="400"></canvas>
    
    <button onclick="logoyuCiz()">✨ DESENİ ÇİZ</button>

    <script>
        // ========================================================
        // 🛠️ 2. GÖREV: OYUN AYARLARI (GEOMETRİ LABORATUVARI)
        // Kendi özgün logonunu yaratmak için buradaki değerleri değiştir!
        // ========================================================
        
        // Şekil Türü: "cizgi" (Doğru parçası) veya "cember" yazabilirsin.
        let anaSekil = "cember"; 
        
        // Tekrar Sayısı: Şekil kaç kere çizilecek? (Örn: 4, 8, 12, 36)
        let tekrarSayisi = 12; 
        
        // Dönme Açısı: Her çizimden sonra kaç derece dönecek? (Örn: 90, 45, 30, 10)
        let donmeAcisi = 30; 
        
        // Logonun Rengi ve Boyutu
        let cizgiRengi = "#f1c40f"; // Sarı
        let sekilBoyutu = 80;       // Çizginin uzunluğu veya çemberin büyüklüğü
        
        // ========================================================
        // ⚠️ DİKKAT: BURADAN AŞAĞISI ÇİZİM MOTORUDUR.
        // ========================================================

        const tuval = document.getElementById("gameCanvas");
        const ctx = tuval.getContext("2d");

        function logoyuCiz() {
            ctx.clearRect(0, 0, tuval.width, tuval.height); // Ekranı temizle
            
            // Çizim merkezini ekranın tam ortasına al
            ctx.translate(tuval.width / 2, tuval.height / 2);
            
            ctx.strokeStyle = cizgiRengi;
            ctx.lineWidth = 3;
            ctx.shadowBlur = 5;
            ctx.shadowColor = cizgiRengi;

            // Dereceyi bilgisayarın anladığı radyan cinsine çeviriyoruz
            let aciRadyan = donmeAcisi * (Math.PI / 180);

            // DÖNGÜ BAŞLIYOR!
            for (let i = 0; i < tekrarSayisi; i++) {
                
                ctx.beginPath();
                
                if (anaSekil === "cizgi") {
                    // Merkezden dışarı doğru bir doğru parçası çiz
                    ctx.moveTo(0, 0);
                    ctx.lineTo(0, -sekilBoyutu);
                } 
                else if (anaSekil === "cember") {
                    // Merkeze tutunarak bir çember çiz
                    ctx.arc(0, -sekilBoyutu/2, sekilBoyutu/2, 0, Math.PI * 2);
                }
                
                ctx.stroke();
                
                // Belirlediğin açı kadar dön!
                ctx.rotate(aciRadyan);
            }
            
            // Ayarları sıfırla ki bir sonraki çizimde ekran kaymasın
            ctx.resetTransform(); 
        }
    </script>
